(defun fdi/tflite-dot-escape (s)
  "Escape S for a Graphviz double-quoted string."
  (apply
   #'concat
   (mapcar
    (lambda (ch)
      (cond
       ;; DOT string escape sequences.
       ((eq ch ?\\) "\\\\")
       ((eq ch ?\") "\\\"")
       ;; Graphviz uses a literal backslash-n for a label line break.
       ((eq ch ?\n) "\\n")
       ((eq ch ?\r) "")
       (t (char-to-string ch))))
    (string-to-list (or s "")))))

(defun fdi/tflite-parse-ops (text)
  "Parse the initial numbered TFLite op listing from TEXT.

Return a list of plists containing:
  :idx :kind :supported :inputs :output :dtype :activation"
  (let ((ops nil)
        (lines (split-string text "\n" t))
        ;; The first listing has global operation numbers.  Stop before the
        ;; later per-partition reports, whose operation indices restart at 0.
        (stop-re "^teflon: ===== subgraph"))
    (catch 'done
      (dolist (line lines)
        (when (string-match-p stop-re line)
          (throw 'done nil))

        ;; Example:
        ;;
        ;;  20 op:  0 ADD v2  🟢 supported in: 174(i8) 183(i8) out: 184(i8)
        ;;
        (when
            (string-match
             (concat
              "^[[:space:]]*\\([0-9]+\\)[[:space:]]+op:"
              "[[:space:]]*[0-9]+[[:space:]]+"
              "\\([^[:space:]]+\\)"
              ".*?"
              "\\(🟢 supported\\|🔴 unsupported\\)"
              ".*in:[[:space:]]*"
              "\\(.*\\)"
              "[[:space:]]+out:[[:space:]]*"
              ;; t184(i8): capture 184 and i8.
              "\\([0-9]+\\)(\\([^)]*\\))"
              "\\(.*\\)$")
             line)
          (let* ((idx        (string-to-number (match-string 1 line)))
                 (kind       (match-string 2 line))
                 (supported  (string= (match-string 3 line)
                                      "🟢 supported"))
                 (input-text (match-string 4 line))
                 (output     (string-to-number (match-string 5 line)))
                 (dtype      (match-string 6 line))
                 (activation (string-trim (match-string 7 line)))
                 (inputs nil)
                 (pos 0))

            ;; Extract all references such as 174(i8), 183(i8), 186(i32).
            ;;
            ;; Group 1: tensor number
            ;; Group 2: type, unused here
            (while (string-match
                    "\\([0-9]+\\)(\\([^)]*\\))"
                    input-text
                    pos)
              (push (string-to-number (match-string 1 input-text))
                    inputs)
              (setq pos (match-end 0)))

            (push (list :idx idx
                        :kind kind
                        :supported supported
                        :inputs (nreverse inputs)
                        :output output
                        :dtype dtype
                        :activation activation)
                  ops)))))
    (nreverse ops)))

(defun fdi/tflite-parse-subgraph-outputs (text)
  "Return all tensors reported as Teflon subgraph outputs in TEXT."
  (let (outputs)
    (dolist (line (split-string text "\n" t))
      ;; Example:
      ;; teflon:   out: 184,252,293
      (when (string-match
             "^teflon:[[:space:]]+out:[[:space:]]*\\(.+\\)$"
             line)
        (dolist (item (split-string (match-string 1 line)
                                    "[,[:space:]]+"
                                    t))
          (when (string-match-p "^[0-9]+$" item)
            (push (string-to-number item) outputs)))))
    (delete-dups outputs)))

(defun fdi/tflite-op-label (op)
  "Return a compact multi-line DOT label for OP."
  (let ((activation (plist-get op :activation)))
    ;; These are real newlines.  fdi/tflite-dot-escape converts them to \n
    ;; for Graphviz after escaping the rest of the string.
    (format "#%d  %s\nout t%d (%s)%s"
            (plist-get op :idx)
            (plist-get op :kind)
            (plist-get op :output)
            (plist-get op :dtype)
            (if (string-empty-p activation)
                ""
              (concat "\n" activation)))))

(defun fdi/tflite-output-path (graph-name extension)
  "Return output path for GRAPH-NAME.EXT next to the current Org file."
  (let* ((base-dir
          (file-name-directory
           (or (buffer-file-name) default-directory)))
         (graph-dir (expand-file-name "graphs" base-dir)))
    (make-directory graph-dir t)
    (expand-file-name
     (format "%s.%s" graph-name extension)
     graph-dir)))

(defun fdi/tflite-write-graph (run-output graph-name)
  "Create a DOT and SVG graph directly from RUN-OUTPUT.

Colours:
  green           supported operation
  red             unsupported operation
  purple outline  Teflon subgraph output
  orange dashed   delegated producer -> CPU consumer
  thick orange    mixed fan-out: delegated producer feeds NPU and CPU"
  (unless (executable-find "dot")
    (user-error "Graphviz 'dot' is not available in exec-path"))

  (let* ((ops (fdi/tflite-parse-ops run-output))
         (subgraph-outs (fdi/tflite-parse-subgraph-outputs run-output))
         ;; tensor ID -> op producing it
         (producer (make-hash-table :test #'eql))
         ;; tensor ID -> ops consuming it
         (consumers (make-hash-table :test #'eql))
         (dotfile (fdi/tflite-output-path graph-name "dot"))
         (svgfile (fdi/tflite-output-path graph-name "svg")))

    (unless ops
      (user-error
       "No initial numbered TFLite operator listing found in run output"))

    (dolist (op ops)
      (puthash (plist-get op :output) op producer)
      (dolist (tensor (plist-get op :inputs))
        (push op (gethash tensor consumers))))

    (with-temp-file dotfile
      (insert "digraph tflite {\n")
      (insert "  graph [rankdir=TB, bgcolor=\"#1e1e1e\", "
              "fontname=\"sans\", pad=\"0.35\", nodesep=\"0.28\", "
              "ranksep=\"0.55\", concentrate=false];\n")
      (insert "  node [shape=box, style=\"rounded,filled\", "
              "fontname=\"sans\", fontsize=10, margin=\"0.10,0.07\", "
              "fontcolor=\"#eeeeee\", color=\"#9a9a9a\"];\n")
      (insert "  edge [fontname=\"sans\", fontsize=8, arrowsize=0.7, "
              "fontcolor=\"#dddddd\", color=\"#9a9a9a\"];\n\n")

      (insert "  label=\"TFLite / Teflon operator data flow\\n"
              "green: supported · red: unsupported · "
              "purple: Teflon subgraph output · "
              "orange: delegated → CPU boundary\";\n")
      (insert "  labelloc=t; fontsize=18; fontcolor=\"#eeeeee\";\n\n")

      ;; Operation nodes.
      (dolist (op ops)
        (let* ((idx (plist-get op :idx))
               (supported (plist-get op :supported))
               (out-tensor (plist-get op :output))
               (subgraph-output-p (member out-tensor subgraph-outs))
               (fillcolor (if supported "#246b45" "#7a3030"))
               (border (if subgraph-output-p "#b67cff" "#9a9a9a"))
               (penwidth (if subgraph-output-p "2.6" "1.0")))
          (insert
           (format
            "  op%d [label=\"%s\", fillcolor=\"%s\", color=\"%s\", penwidth=%s];\n"
            idx
            (fdi/tflite-dot-escape (fdi/tflite-op-label op))
            fillcolor
            border
            penwidth))))

      (insert "\n")

      ;; Sort tensor IDs to make DOT output stable between runs.
      (let (tensor-ids)
        (maphash
         (lambda (tensor _producer-op)
           (push tensor tensor-ids))
         producer)

        (dolist (tensor (sort tensor-ids #'<))
          (let* ((producer-op (gethash tensor producer))
                 (producer-supported
                  (plist-get producer-op :supported))
                 (all-consumers
                  (sort
                   (delete-dups
                    (copy-sequence (gethash tensor consumers)))
                   (lambda (a b)
                     (< (plist-get a :idx)
                        (plist-get b :idx)))))
                 (has-supported-consumer
                  (cl-some
                   (lambda (op) (plist-get op :supported))
                   all-consumers))
                 (has-unsupported-consumer
                  (cl-some
                   (lambda (op) (not (plist-get op :supported)))
                   all-consumers))
                 ;; The interesting case: e.g. t184/t252 in your graph.
                 ;; A native-layout delegate result continues on NPU but must
                 ;; also be exposed in ordinary TFLite logical layout to CPU.
                 (mixed-fanout
                  (and producer-supported
                       has-supported-consumer
                       has-unsupported-consumer)))

            (dolist (consumer all-consumers)
              (let* ((consumer-supported
                      (plist-get consumer :supported))
                     (delegated-to-cpu
                      (and producer-supported
                           (not consumer-supported)))
                     (attrs
                      (cond
                       (mixed-fanout
                        (list
                         "color=\"#ffad42\""
                         "fontcolor=\"#ffcf86\""
                         "penwidth=2.8"
                         (format
                          "label=\"t%d%s\""
                          tensor
                          (if delegated-to-cpu
                              "  export → CPU"
                            "  stays delegated"))))

                       (delegated-to-cpu
                        (list
                         "color=\"#ffad42\""
                         "fontcolor=\"#ffcf86\""
                         "style=dashed"
                         "penwidth=2.0"
                         (format
                          "label=\"t%d  export → CPU\""
                          tensor)))

                       ((member tensor subgraph-outs)
                        (list
                         "color=\"#b67cff\""
                         "fontcolor=\"#d3b5ff\""
                         "penwidth=1.8"
                         (format
                          "label=\"t%d  subgraph output\""
                          tensor))))))

                (insert
                 (format
                  "  op%d -> op%d%s;\n"
                  (plist-get producer-op :idx)
                  (plist-get consumer :idx)
                  (if attrs
                      (format " [%s]" (string-join attrs ", "))
                    ""))))))))

      (insert "\n")
      (insert "  subgraph cluster_legend {\n")
      (insert "    label=\"Legend\"; color=\"#666666\"; "
              "fontcolor=\"#dddddd\"; fontsize=10;\n")
      (insert "    key_supported [label=\"supported\", fillcolor=\"#246b45\"];\n")
      (insert "    key_unsupported [label=\"unsupported\", fillcolor=\"#7a3030\"];\n")
      (insert "    key_output [label=\"Teflon subgraph output\", "
              "fillcolor=\"#333333\", color=\"#b67cff\", penwidth=2.6];\n")
      (insert "  }\n")
      (insert "}\n"))

    (let ((status
           (call-process "dot" nil nil nil
                         "-Tsvg" dotfile "-o" svgfile)))
      (unless (zerop status)
        (error "Graphviz failed while rendering %s" dotfile)))

    svgfile))
